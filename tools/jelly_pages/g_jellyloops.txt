/* ============ Jelly Loops: oynanabilir iki bulmaca ailesi ============
   Kurallar oyunun kendi kodundan (loops.dart). Halka çiz: sayılı hücrenin çevresinde o kadar ip olsun, ipler TEK kapalı halka olsun;
   bir kenara dokunmak ip → X → boş döngüsüdür. Köprü kur: ada sayısı kadar köprü, köprüler kesişmez, tüm adalar bağlanır;
   ikili köprü açıksa bir bağ iki kez dokununca çift olur. Bulmacalar tek çözümlü. */
(function(){
  'use strict';
  var JG=window.JG,$=JG.$,$$=JG.$$,JX=window.JX;
  var P=window.JX_PUZZLES; if(!P||!P.length) return;
  var NS='http://www.w3.org/2000/svg', S=100, PAD=50;
  var host=$('#board'), pal=$('#lp-pal');
  var cur=0, pz, won=false, solved=P.map(function(){return false;});
  var E={}, B=[];           /* E: 'h,r,c' -> 0|1|2 ; B: köprü sayıları */
  var NAMES=[['3×3 loop','3×3 halka'],['4×4 loop','4×4 halka'],['5×5 loop','5×5 halka'],['Bridges','Köprüler'],['Double bridges','Çift köprü']];
  var el=function(tag,attrs,parent){ var e=document.createElementNS(NS,tag); for(var k in attrs) e.setAttribute(k,attrs[k]); if(parent) parent.appendChild(e); return e; };
  var ek=function(e){ return e[0]+','+e[1]+','+e[2]; };

  function load(i){
    cur=i; pz=P[i]; won=false; E={}; B=[]; $('#win').classList.remove('on'); pal.src='img/s/loops_hero_idle.webp'; JX.hint('','');
    host.innerHTML=''; $$('#board svg').forEach(function(x){x.remove();});
    var lab=P.map(function(p,k){ return '<span class="en i">'+NAMES[k][0]+'</span><span class="tr i">'+NAMES[k][1]+'</span>'; });
    JX.picker($('#picker'),lab,cur,solved,load);
    if(pz.type==='line') buildLine(); else buildBridge();
    labels();
  }
  function labels(){
    var line=pz.type==='line';
    $('#m-a-l').innerHTML=line?'<span class="en i">Numbers happy</span><span class="tr i">Mutlu sayı</span> ':'<span class="en i">Islands happy</span><span class="tr i">Mutlu ada</span> ';
    $('#m-b-l').innerHTML=line?'<span class="en i">Threads</span><span class="tr i">İp</span> ':'<span class="en i">Bridges</span><span class="tr i">Köprü</span> ';
  }

  /* ------------------------- halka çiz ------------------------- */
  function buildLine(){
    var n=pz.n, W=n*S+PAD*2; var svg=el('svg',{viewBox:'0 0 '+W+' '+W,'class':'lp-svg'},host); host.appendChild(svg);
    var cellG=el('g',{},svg), cellImgs={};
    pz.clues.forEach(function(c){ var im=el('image',{href:'img/s/loops_cell.webp',x:PAD+c[1]*S+4,y:PAD+c[0]*S+4,width:S-8,height:S-8},cellG); cellImgs[c[0]+','+c[1]]=im; });
    var all=[]; for(var r=0;r<=n;r++) for(var c=0;c<n;c++) all.push(['h',r,c]); for(r=0;r<n;r++) for(c=0;c<=n;c++) all.push(['v',r,c]);
    var thG=el('g',{},svg), txG=el('g',{},svg), hitG=el('g',{},svg), dotG=el('g',{},svg), xG=el('g',{},svg);
    pz._cl={}; pz.clues.forEach(function(c){ pz._cl[c[0]+','+c[1]]=el('text',{x:PAD+c[1]*S+S/2,y:PAD+c[0]*S+S/2+2,'class':'cl'},txG); pz._cl[c[0]+','+c[1]].textContent=c[2]; });
    for(r=0;r<=n;r++) for(c=0;c<=n;c++) el('image',{href:'img/s/loops_dot.webp',x:PAD+c*S-13,y:PAD+r*S-13,width:26,height:26},dotG);
    pz._th={}; pz._x={};
    all.forEach(function(e){
      var x1=PAD+e[2]*S, y1=PAD+e[1]*S, x2=e[0]==='h'?x1+S:x1, y2=e[0]==='v'?y1+S:y1;
      var g=el('g',{},thG); var o=el('line',{x1:x1,y1:y1,x2:x2,y2:y2,'class':'th',opacity:0},g), i2=el('line',{x1:x1,y1:y1,x2:x2,y2:y2,'class':'thi',opacity:0},g); pz._th[ek(e)]=[o,i2];
      var xi=el('image',{href:'img/s/loops_edge_x.webp',x:(x1+x2)/2-14,y:(y1+y2)/2-14,width:28,height:28,opacity:0,'pointer-events':'none'},xG); pz._x[ek(e)]=xi;
      var h=el('line',{x1:x1+14*(e[0]==='h'),y1:y1+14*(e[0]==='v'),x2:x2-14*(e[0]==='h'),y2:y2-14*(e[0]==='v'),'class':'eh'},hitG);
      h.addEventListener('pointerdown',function(ev){ ev.preventDefault(); toggleEdge(e); });
    });
    renderLine();
  }
  function toggleEdge(e){ if(won) return; var k=ek(e); E[k]=((E[k]||0)+1)%3; renderLine(); }
  function cellCount(r,c,E2){ var n=0; [['h',r,c],['h',r+1,c],['v',r,c],['v',r,c+1]].forEach(function(e){ if(E2[ek(e)]===1) n++; }); return n; }
  function renderLine(){
    var n=pz.n, threads=0;
    Object.keys(pz._th).forEach(function(k){ var v=E[k]||0; pz._th[k][0].setAttribute('opacity',v===1?1:0); pz._th[k][1].setAttribute('opacity',v===1?1:0); pz._x[k].setAttribute('opacity',v===2?.9:0); if(v===1) threads++; });
    var happy=0;
    pz.clues.forEach(function(c){ var k=c[0]+','+c[1], cnt=cellCount(c[0],c[1],E), t=pz._cl[k];
      t.setAttribute('class','cl'+(cnt===c[2]?' ok':cnt>c[2]?' bad':'')); if(cnt===c[2]) happy++; });
    $('#m-a').textContent=happy+'/'+pz.clues.length; $('#m-b').textContent=threads;
    if(!won){ var sol={}; pz.solution.forEach(function(e){ sol[ek(e)]=1; }); var ok=true, cnt=0; Object.keys(E).forEach(function(k){ if(E[k]===1){ cnt++; if(!sol[k]) ok=false; } }); if(ok && cnt===pz.solution.length) win(); }
  }

  /* ------------------------- köprü kur ------------------------- */
  function buildBridge(){
    var n=pz.n, W=(n-1)*S+PAD*2+40; var svg=el('svg',{viewBox:'0 0 '+W+' '+W,'class':'lp-svg'},host);
    var ox=PAD+20, oy=PAD+20;
    pz._pos=pz.islands.map(function(s){ return [ox+s[1]*S, oy+s[0]*S]; });
    pz._link=[]; B=pz.links.map(function(){return 0;});
    var lg=el('g',{},svg), ig=el('g',{},svg), hg=el('g',{},svg);
    pz.links.forEach(function(l,i){
      var a=pz._pos[l[0]], b=pz._pos[l[1]], horiz=pz.islands[l[0]][0]===pz.islands[l[1]][0];
      var g=el('g',{},lg), shapes=[];
      for(var k=0;k<2;k++){ var off=(k===0?-7:7); var o=el('line',{x1:a[0]+(horiz?0:off),y1:a[1]+(horiz?off:0),x2:b[0]+(horiz?0:off),y2:b[1]+(horiz?off:0),'class':'bo',stroke:'#E8458E','stroke-width':8,opacity:0},g); var o2=el('line',{x1:o.getAttribute('x1'),y1:o.getAttribute('y1'),x2:o.getAttribute('x2'),y2:o.getAttribute('y2'),'class':'bi','stroke-width':3,opacity:0},g); shapes.push([o,o2]); }
      var c1=el('line',{x1:a[0],y1:a[1],x2:b[0],y2:b[1],'class':'bo',stroke:'#E8458E','stroke-width':9,opacity:0},g), c2=el('line',{x1:a[0],y1:a[1],x2:b[0],y2:b[1],'class':'bi','stroke-width':3,opacity:0},g);
      pz._link.push({single:[c1,c2],dbl:shapes});
      var h=el('line',{x1:a[0]+(horiz?34:0),y1:a[1]+(horiz?0:34),x2:b[0]-(horiz?34:0),y2:b[1]-(horiz?0:34),'class':'bz'},hg);
      h.addEventListener('pointerdown',function(ev){ ev.preventDefault(); cycle(i); });
    });
    pz._isl=pz.islands.map(function(s,i){ var p=pz._pos[i]; var im=el('image',{href:'img/s/loops_island.webp',x:p[0]-40,y:p[1]-40,width:80,height:80},ig); var t=el('text',{x:p[0],y:p[1]+2,'class':'nm'},ig); t.textContent=s[2]; return {im:im,t:t}; });
    renderBridge();
  }
  function crosses(i,j){
    var a=pz.islands[pz.links[i][0]], a2=pz.islands[pz.links[i][1]], b=pz.islands[pz.links[j][0]], b2=pz.islands[pz.links[j][1]];
    var h1=a[0]===a2[0], h2=b[0]===b2[0]; if(h1===h2) return false;
    var H=h1?[a,a2]:[b,b2], V=h1?[b,b2]:[a,a2];
    var hr=H[0][0], hc0=Math.min(H[0][1],H[1][1]), hc1=Math.max(H[0][1],H[1][1]), vc=V[0][1], vr0=Math.min(V[0][0],V[1][0]), vr1=Math.max(V[0][0],V[1][0]);
    return hc0<vc && vc<hc1 && vr0<hr && hr<vr1;
  }
  function cycle(i){
    if(won) return; var max=pz.double?2:1, nv=(B[i]+1)%(max+1);
    if(nv>0 && B[i]===0){ for(var j=0;j<B.length;j++){ if(j!==i && B[j]>0 && crosses(i,j)){ var g=pz._link[i].single[0]; JX.hint('Bridges cannot cross.','Köprüler kesişemez.'); return; } } }
    B[i]=nv; JX.hint('',''); renderBridge();
  }
  function degs(){ var d=pz.islands.map(function(){return 0;}); pz.links.forEach(function(l,i){ d[l[0]]+=B[i]; d[l[1]]+=B[i]; }); return d; }
  function renderBridge(){
    pz._link.forEach(function(o,i){ o.single.forEach(function(x){ x.setAttribute('opacity',B[i]===1?1:0); }); o.dbl.forEach(function(p){ p.forEach(function(x){ x.setAttribute('opacity',B[i]===2?1:0); }); }); });
    var d=degs(), happy=0, total=0;
    pz.islands.forEach(function(s,i){ var ok=d[i]===s[2], bad=d[i]>s[2]; if(ok) happy++;
      pz._isl[i].im.setAttribute('href','img/s/loops_island'+(ok?'_ok':bad?'_err':'')+'.webp'); pz._isl[i].t.setAttribute('class','nm'+(ok?' ok':bad?' bad':'')); });
    B.forEach(function(b){ total+=b; });
    $('#m-a').textContent=happy+'/'+pz.islands.length; $('#m-b').textContent=total;
    if(!won && happy===pz.islands.length){
      var seen={0:1}, st=[0]; while(st.length){ var x=st.pop(); pz.links.forEach(function(l,i){ if(B[i]>0){ var y=l[0]===x?l[1]:l[1]===x?l[0]:-1; if(y>=0&&!seen[y]){ seen[y]=1; st.push(y); } } }); }
      if(Object.keys(seen).length===pz.islands.length) win();
    }
  }

  function win(){
    won=true; solved[cur]=true; pal.src='img/s/loops_hero_win.webp'; var svg=host.querySelector('svg'); svg.classList.add('won');
    JX.picker($('#picker'),P.map(function(p,k){ return '<span class="en i">'+NAMES[k][0]+'</span><span class="tr i">'+NAMES[k][1]+'</span>'; }),cur,solved,load);
    JX.win(['fx_star','fx_heart','loops_pearl_white','loops_pearl_black']);
  }
  $('#b-reset').addEventListener('click',function(){ load(cur); });
  $('#b-next').addEventListener('click',function(){ load((cur+1)%P.length); });
  $('#b-hint').addEventListener('click',function(){
    if(won) return;
    if(pz.type==='line'){
      var sol={}; pz.solution.forEach(function(e){ sol[ek(e)]=1; }); var k;
      for(k in E){ if(E[k]===1 && !sol[k]){ E[k]=0; JX.hint('A thread that does not belong was removed.','Yerinde olmayan bir ip kaldırıldı.'); renderLine(); return; } }
      for(var i=0;i<pz.solution.length;i++){ k=ek(pz.solution[i]); if(E[k]!==1){ E[k]=1; JX.hint('One missing thread was added.','Eksik bir ip eklendi.'); renderLine(); return; } }
    } else {
      for(var j=0;j<B.length;j++){ if(B[j]!==pz.solution[j]){ for(var q=0;q<B.length;q++){ if(q!==j && B[q]>0 && pz.solution[j]>0 && crosses(j,q)) B[q]=0; } B[j]=pz.solution[j]; JX.hint('One bridge was fixed.','Bir köprü düzeltildi.'); renderBridge(); return; } }
    }
  });
  load(0);
})();
