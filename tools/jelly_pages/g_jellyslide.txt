/* ============ Jelly Slide: oynanabilir buz bulmacası ============
   Kurallar oyunun kendi kodundan (slide.dart): penguen bir yöne kayar, kaya ya da kenar durdurana kadar; kar yığınına girince durur;
   balıklar üzerinden geçilirken toplanır ve hepsi toplanmadan çıkış açılmaz; ışınlanma deliğine girince eşinden AYNI yönde devam eder;
   çıkış deliğine (balıklar tamamsa) girince bölüm biter. En az hamle bulmacadaki "en iyi" sayıdır. */
(function(){
  'use strict';
  var JG=window.JG,$=JG.$,$$=JG.$$,JX=window.JX;
  var P=window.JX_PUZZLES; if(!P||!P.length) return;
  var board=$('#board'), D={u:[-1,0],d:[1,0],l:[0,-1],r:[0,1]};
  var cur=0, pz, n, pos, fishLeft, moves, hist, busy, won, penEl, fishEls, solved=P.map(function(){return false;}), exitEl;
  var NAMES=[['Rocks','Kayalar'],['Snow piles','Kar yığınları'],['Fish','Balıklar'],['Warp holes','Işınlanma']];
  var key=function(q){ return q[0]+','+q[1]; };

  function setOf(a){ var m={}; a.forEach(function(q){ m[key(q)]=1; }); return m; }
  function load(i){
    cur=i; pz=P[i]; n=pz.n; pos=pz.start.slice(); moves=0; hist=[]; busy=false; won=false; fishLeft=pz.fish.map(function(){return true;});
    pz._rock=setOf(pz.rocks); pz._snow=setOf(pz.snow); pz._warp={}; pz.warps.forEach(function(w){ pz._warp[key(w[0])]=w[1]; pz._warp[key(w[1])]=w[0]; });
    $('#win').classList.remove('on'); JX.hint('','');
    JX.picker($('#picker'),P.map(function(p,k){ return '<span class="en i">'+NAMES[k][0]+'</span><span class="tr i">'+NAMES[k][1]+'</span>'; }),cur,solved,load);
    board.innerHTML=''; board.style.setProperty('--n',n);
    for(var r=0;r<n;r++) for(var c=0;c<n;c++){ var t=document.createElement('i'); t.className='sl-t'; t.style.cssText='left:'+(c/n*100)+'%;top:'+(r/n*100)+'%;background-image:url(img/s/slide_ice_'+((r+c)%2?'b':'a')+'.webp)'; board.appendChild(t); }
    function obj(q,cls,src){ var d=document.createElement('div'); d.className='sl-o '+cls; d.style.cssText='left:'+(q[1]/n*100)+'%;top:'+(q[0]/n*100)+'%'; d.innerHTML='<img src="img/s/'+src+'.webp" alt="">'; board.appendChild(d); return d; }
    pz.snow.forEach(function(q){ obj(q,'snow','slide_snow'); });
    pz.rocks.forEach(function(q,k){ obj(q,'rock','slide_rock_'+(k%3)); });
    pz.warps.forEach(function(w,k){ obj(w[0],'warp','slide_warp_'+(k%3)); obj(w[1],'warp','slide_warp_'+(k%3)); });
    exitEl=obj(pz.exit,'exit',pz.fish.length?'slide_exit_closed':'slide_exit_open');
    fishEls=pz.fish.map(function(q,k){ return obj(q,'fish','slide_fish_'+(k%2)); });
    penEl=document.createElement('div'); penEl.className='sl-pen'; penEl.innerHTML='<img src="img/s/slide_penguin_idle.webp" alt="">'; place(pos,0); board.appendChild(penEl);
    $('#m-fish-w').style.display=pz.fish.length?'':'none'; stats();
  }
  function place(q,dur){ penEl.style.setProperty('--dur',dur+'s'); penEl.style.left=(q[1]/n*100)+'%'; penEl.style.top=(q[0]/n*100)+'%'; }
  function stats(){ $('#m-moves').textContent=moves; $('#m-par').textContent=pz.par; $('#m-fish').textContent=(pz.fish.length-fishLeft.filter(Boolean).length)+'/'+pz.fish.length; }
  /* bir hamlenin sonucu: oyun kuralıyla aynı */
  function run(from,d,fl){
    var r=from[0], c=from[1], steps=0, got=[], dv=D[d], done=false, path=[];
    fl=fl.slice();
    while(true){
      var nr=r+dv[0], nc=c+dv[1];
      if(nr<0||nr>=n||nc<0||nc>=n||pz._rock[nr+','+nc]) break;
      r=nr; c=nc; steps++; path.push([r,c]); if(steps>4*n) break;
      pz.fish.forEach(function(f,k){ if(fl[k]&&f[0]===r&&f[1]===c){ fl[k]=false; got.push(k); } });
      var w=pz._warp[r+','+c]; if(w){ r=w[0]; c=w[1]; path.push([r,c]); }
      if(r===pz.exit[0]&&c===pz.exit[1]&&fl.every(function(x){return !x;})){ done=true; break; }
      if(pz._snow[r+','+c]) break;
      if(steps>4*n) break;
    }
    return {to:[r,c],steps:steps,got:got,done:done,fl:fl,path:path};
  }
  function go(d,silent){
    if(busy||won) return; var res=run(pos,d,fishLeft);
    if(!res.steps){ penEl.classList.remove('bonk'); void penEl.offsetWidth; penEl.classList.add('bonk'); return; }
    busy=true; hist.push({pos:pos.slice(),fl:fishLeft.slice(),moves:moves}); moves++;
    var img=penEl.querySelector('img'); img.src='img/s/slide_penguin_'+({u:'up',d:'down',l:'left',r:'right'})[d]+'.webp';
    var dur=Math.min(.9,.16+res.steps*.09); place(res.to,dur);
    setTimeout(function(){
      pos=res.to; fishLeft=res.fl; res.got.forEach(function(k){ fishEls[k].classList.add('gone'); });
      if(pz.fish.length && fishLeft.every(function(x){return !x;})) exitEl.querySelector('img').src='img/s/slide_exit_open.webp';
      stats();
      if(res.done){ won=true; solved[cur]=true; penEl.classList.add('dive'); setTimeout(function(){ JX.picker($('#picker'),P.map(function(p,k){ return '<span class="en i">'+NAMES[k][0]+'</span><span class="tr i">'+NAMES[k][1]+'</span>'; }),cur,solved,load); JX.win(['fx_star','fx_heart','fx_flake','fx_glint']); },450); }
      else img.src='img/s/slide_penguin_idle.webp';
      busy=false;
    },dur*1000+30);
  }
  /* kaydırma: işaretçi, ok tuşları, ekran okları */
  var sx=0, sy=0, down=false;
  board.addEventListener('pointerdown',function(e){ down=true; sx=e.clientX; sy=e.clientY; try{ board.setPointerCapture(e.pointerId); }catch(x){} e.preventDefault(); });
  board.addEventListener('pointerup',function(e){
    if(!down) return; down=false; var dx=e.clientX-sx, dy=e.clientY-sy; if(Math.max(Math.abs(dx),Math.abs(dy))<18) return;
    go(Math.abs(dx)>Math.abs(dy)?(dx>0?'r':'l'):(dy>0?'d':'u'));
  });
  $$('#pad button').forEach(function(b){ b.addEventListener('click',function(){ go(b.getAttribute('data-d')); }); });
  document.addEventListener('keydown',function(e){
    var m={ArrowUp:'u',ArrowDown:'d',ArrowLeft:'l',ArrowRight:'r'}[e.key]; if(!m) return;
    var rc=board.getBoundingClientRect(); if(rc.bottom<0||rc.top>innerHeight) return;
    if(e.target.closest&&e.target.closest('input,textarea,select')) return; e.preventDefault(); go(m);
  });
  $('#b-reset').addEventListener('click',function(){ load(cur); });
  $('#b-next').addEventListener('click',function(){ load((cur+1)%P.length); });
  $('#b-undo').addEventListener('click',function(){
    if(busy||!hist.length) return; var h=hist.pop(); pos=h.pos; fishLeft=h.fl; moves=h.moves; won=false; penEl.classList.remove('dive'); $('#win').classList.remove('on');
    fishEls.forEach(function(el,k){ el.classList.toggle('gone',!fishLeft[k]); }); exitEl.querySelector('img').src='img/s/slide_exit_'+(fishLeft.some(Boolean)?'closed':'open')+'.webp'; place(pos,.2); stats();
  });
  /* ipucu: şu andan çıkışa en kısa yol (BFS) */
  function best(){
    var start=[pos[0],pos[1],fishLeft.map(function(x){return x?1:0;}).join('')], seen={}, q=[{s:start,f:null}], h=0; seen[start.join('|')]=1;
    while(h<q.length&&q.length<20000){ var e=q[h++];
      for(var d in D){ var fl=e.s[2].split('').map(function(x){return x==='1';}); var r=run([e.s[0],e.s[1]],d,fl); if(!r.steps) continue;
        if(r.done) return e.f||d; var k=[r.to[0],r.to[1],r.fl.map(function(x){return x?1:0;}).join('')], ks=k.join('|'); if(seen[ks]) continue; seen[ks]=1; q.push({s:k,f:e.f||d}); } }
    return null;
  }
  $('#b-hint').addEventListener('click',function(){
    if(busy||won) return; var d=best(); if(!d){ JX.hint('No way out from here. Try Undo or Start over.','Buradan çıkış yok. Geri al ya da baştan başla.'); return; }
    var b=$('#pad [data-d="'+d+'"]'); b.classList.add('hl2'); setTimeout(function(){ b.classList.remove('hl2'); },2200);
    JX.hint('Try sliding that way.','Şu yöne kaymayı dene.');
  });
  load(0);
})();
