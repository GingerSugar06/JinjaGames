/* ============ Jelly Seek: oynanabilir bulmaca ============
   Kurallar oyunun kendi kodundan (seek.dart): açılan karonun sayısı ona değen sekiz komşudaki sürprizleri sayar, sayısı 0 olan
   karo komşularını da açar; fener (hep güvenli) bütün SATIRI, dürbün (hep güvenli) iki kare uzaklıktaki 5x5 bölgeyi sayar.
   Bulmacalar tahminsiz: oyunun çözücüsüne benzer çıkarımla (sayım ve küme farkı) tamamen çözülebilir. */
(function(){
  'use strict';
  var JG=window.JG,$=JG.$,$$=JG.$$,JX=window.JX;
  var P=window.JX_PUZZLES; if(!P||!P.length) return;
  var board=$('#board'), hero=$('#sk-hero');
  var cur=0, pz, R, C, mines, kinds, opened, flags, tiles, over, won, flagMode=false, solved=P.map(function(){return false;});
  var NAMES=[['Starter 5×5','Başlangıç 5×5'],['Classic 6×6','Klasik 6×6'],['Lanterns','Fenerler'],['Spyglass','Dürbün']];

  function nb(i){ var r=Math.floor(i/C), c=i%C, o=[]; for(var a=-1;a<=1;a++) for(var b=-1;b<=1;b++){ if((a||b)&&r+a>=0&&r+a<R&&c+b>=0&&c+b<C) o.push((r+a)*C+c+b); } return o; }
  function region(i,k){
    var r=Math.floor(i/C), c=i%C, o=[];
    if(k==='l'){ for(var cc=0;cc<C;cc++) if(cc!==c) o.push(r*C+cc); return o; }
    if(k==='s'){ for(var a=-2;a<=2;a++) for(var b=-2;b<=2;b++){ if((a||b)&&r+a>=0&&r+a<R&&c+b>=0&&c+b<C) o.push((r+a)*C+c+b); } return o; }
    return nb(i);
  }
  function countAt(i){ return region(i,kinds[i]||'n').filter(function(j){ return mines[j]; }).length; }

  function load(i){
    cur=i; pz=P[i]; R=pz.rows; C=pz.cols; mines={}; pz.mines.forEach(function(m){ mines[m]=1; }); kinds={}; Object.keys(pz.kinds).forEach(function(k){ kinds[k]=pz.kinds[k]; });
    opened={}; flags={}; over=false; won=false; $('#win').classList.remove('on'); hero.src='img/s/seek_hero_idle.webp'; JX.hint('','');
    JX.picker($('#picker'),P.map(function(p,k){ return '<span class="en i">'+NAMES[k][0]+'</span><span class="tr i">'+NAMES[k][1]+'</span>'; }),cur,solved,load);
    board.style.setProperty('--cols',C); board.innerHTML=''; tiles=[];
    for(var k=0;k<R*C;k++){ var t=document.createElement('button'); t.type='button'; t.className='sk'+(((Math.floor(k/C)+k%C)%2)?' b':''); t.setAttribute('aria-label','tile '+(k+1));
      (function(k){ t.addEventListener('click',function(){ press(k); }); t.addEventListener('contextmenu',function(e){ e.preventDefault(); toggleFlag(k); });
        var lp=null; t.addEventListener('pointerdown',function(e){ if(e.pointerType==='touch'){ lp=setTimeout(function(){ lp=null; t.__lp=1; toggleFlag(k); },480); } });
        ['pointerup','pointerleave','pointercancel'].forEach(function(ev){ t.addEventListener(ev,function(){ if(lp){ clearTimeout(lp); lp=null; } }); });
        t.addEventListener('click',function(e){ if(t.__lp){ t.__lp=0; e.stopImmediatePropagation(); } },true); })(k);
      board.appendChild(t); tiles.push(t); }
    Object.keys(kinds).forEach(function(k){ reveal(+k); });
    openCell(pz.start); render();
  }
  function openCell(i){
    var st=[i];
    while(st.length){ var x=st.pop(); if(opened[x]||mines[x]) continue; opened[x]=1; delete flags[x];
      if(!kinds[x] && countAt(x)===0){ nb(x).forEach(function(y){ if(!opened[y]) st.push(y); }); } }
  }
  function reveal(i){ opened[i]=1; }
  function render(){
    var left=0;
    tiles.forEach(function(t,i){
      var o=opened[i], k=kinds[i], h='';
      t.className='sk'+(((Math.floor(i/C)+i%C)%2)?' b':'')+(o?' op':'')+(k==='l'?' lt':'')+(k==='s'?' sy':'');
      if(o){ var cnt=countAt(i);
        if(k==='l') h='<img class="sp" src="img/s/seek_lantern.webp" alt=""><span class="nb">'+cnt+'</span>';
        else if(k==='s') h='<img class="sp" src="img/s/seek_hero_search.webp" alt=""><span class="nb">'+cnt+'</span>';
        else h=cnt?'<span class="nb n'+cnt+'">'+cnt+'</span>':'';
      } else { if(!mines[i]) left++; if(flags[i]) h='<img class="fg" src="img/s/seek_flag.webp" alt="">'; }
      if(over && mines[i]) { h='<img class="sp" src="img/s/seek_surprise.webp" alt="">'; t.classList.add('op'); }
      t.innerHTML=h;
    });
    $('#m-left').textContent=left; $('#m-flag').textContent=Object.keys(flags).length+'/'+pz.mines.length;
    if(!over && !won && left===0) win();
  }
  function press(i){
    if(over||won||opened[i]) return;
    if(flagMode){ toggleFlag(i); return; }
    if(flags[i]){ return; }
    if(mines[i]){ over=true; hero.src='img/s/seek_hero_scared.webp'; render(); JX.hint('A sour jelly woke up! That tile was a surprise. Start over and use the numbers.','Ekşi bir jöle uyandı! O karo bir sürprizdi. Baştan başla ve sayılara güven.'); setTimeout(function(){ if(over) load(cur); },2600); return; }
    openCell(i); hero.src='img/s/seek_hero_happy.webp'; setTimeout(function(){ if(!over&&!won) hero.src='img/s/seek_hero_idle.webp'; },600); render();
  }
  function toggleFlag(i){ if(over||won||opened[i]) return; if(flags[i]) delete flags[i]; else flags[i]=1; render(); }
  function win(){
    won=true; solved[cur]=true; hero.src='img/s/seek_hero_win.webp';
    JX.picker($('#picker'),P.map(function(p,k){ return '<span class="en i">'+NAMES[k][0]+'</span><span class="tr i">'+NAMES[k][1]+'</span>'; }),cur,solved,load);
    JX.win(['fx_star','fx_heart','seek_sweet_0','seek_sweet_1','seek_sweet_3']);
  }
  /* ipucu: sayımlardan kesin güvenli bir karo bul (oyundaki ipucu gibi, mantıkla emin olunan bir karoyu gösterir) */
  function safeTile(){
    var N=R*C, mk={}, sf={}, prog=true, guard=0;
    while(prog && guard++<40){
      prog=false; var cons=[];
      Object.keys(opened).forEach(function(x){ x=+x; var reg=region(x,kinds[x]||'n'), cnt=countAt(x), unk=reg.filter(function(y){ return !opened[y]&&!mk[y]; }), rem=cnt-reg.filter(function(y){ return mk[y]; }).length; if(unk.length) cons.push([unk,rem]); });
      cons.forEach(function(c){ if(c[1]===0) c[0].forEach(function(y){ if(!sf[y]){ sf[y]=1; prog=true; } }); else if(c[1]===c[0].length) c[0].forEach(function(y){ if(!mk[y]){ mk[y]=1; prog=true; } }); });
      cons.forEach(function(a){ cons.forEach(function(b){ if(a===b) return; var sub=a[0].every(function(y){ return b[0].indexOf(y)>-1; }); if(sub&&a[0].length<b[0].length){ var d=b[0].filter(function(y){ return a[0].indexOf(y)<0; }), dr=b[1]-a[1]; if(dr===0) d.forEach(function(y){ if(!sf[y]){ sf[y]=1; prog=true; } }); else if(dr===d.length) d.forEach(function(y){ if(!mk[y]){ mk[y]=1; prog=true; } }); } }); });
    }
    for(var k in sf){ if(!opened[k]) return +k; } return -1;
  }
  $('#b-flag').addEventListener('click',function(){ flagMode=!flagMode; this.classList.toggle('on',flagMode); });
  $('#b-reset').addEventListener('click',function(){ load(cur); });
  $('#b-next').addEventListener('click',function(){ load((cur+1)%P.length); });
  $('#b-hint').addEventListener('click',function(){
    if(over||won) return; var s=safeTile();
    if(s<0){ JX.hint('Nothing is certain from the numbers alone right now. Look at the numbers again.','Şu an yalnız sayılardan kesin bir şey çıkmıyor. Sayılara yeniden bak.'); return; }
    tiles[s].classList.add('hint'); setTimeout(function(){ tiles[s].classList.remove('hint'); },3000);
    JX.hint('The glowing tile is certainly safe. Why? Read the numbers around it.','Parlayan karo kesinlikle güvenli. Neden? Çevresindeki sayıları oku.');
  });
  load(0);
})();
