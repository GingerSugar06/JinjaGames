/* ============ Jelly Jars: oynanabilir bulmaca ============
   Kurallar oyunun kendi kodundan (jars.dart): üstteki aynı renkli katmanlar aynı renge ya da boş kavanoza dökülür,
   yer yoksa dökülmez; kilitli kavanoza yalnız kilidin rengi dökülür; gizli katmanlar '?' görünür, üstleri
   boşalınca açılır. Her kavanoz tek renk (ya da boş) olunca bölüm biter. */
(function(){
  'use strict';
  var JG=window.JG,$=JG.$,$$=JG.$$,JX=window.JX;
  var P=window.JX_PUZZLES; if(!P||!P.length) return;
  var CAP=4, board=$('#board'), mascot=$('#jb-mascot');
  var cur=0, st, locks, hidden, sel=-1, hist=[], moves=0, par=0, solved=P.map(function(){return false;}), busy=false, won=false;
  var NAMES=[['Classic','Klasik'],['Hidden layers','Gizli katman'],['Locked jar','Kilitli kavanoz']];

  function copy(s){ return s.map(function(j){return j.slice();}); }
  function top(j){ return j[j.length-1]; }
  function runLen(j){ var c=top(j), n=0; while(n<j.length && j[j.length-1-n]===c) n++; return n; }
  function canPour(s,a,b){
    if(a===b||!s[a].length) return 0; var c=top(s[a]);
    if(locks[b]!==null && locks[b]!==c) return 0;
    if(s[b].length && top(s[b])!==c) return 0;
    var room=CAP-s[b].length; if(room<=0) return 0;
    return Math.min(runLen(s[a]),room);
  }
  function isDone(s){ return s.every(function(j){ return !j.length || (j.length===CAP && j.every(function(x){return x===j[0];})); }); }
  function key(s){ return s.map(function(j){return j.join('');}).join('|'); }
  function solve(s0){
    var seen={}, q=[{s:s0,f:null}], h=0; seen[key(s0)]=1;
    while(h<q.length && q.length<60000){
      var e=q[h++]; if(isDone(e.s)) return e.f;
      for(var a=0;a<e.s.length;a++) for(var b=0;b<e.s.length;b++){
        var k=canPour(e.s,a,b); if(!k) continue;
        if(!e.s[b].length && runLen(e.s[a])===e.s[a].length && locks[b]===null) continue;
        var n=copy(e.s), c=top(n[a]); for(var i=0;i<k;i++){ n[a].pop(); n[b].push(c); }
        var kk=key(n); if(seen[kk]) continue; seen[kk]=1; q.push({s:n,f:e.f||[a,b]});
      }
    }
    return null;
  }

  function render(newJar){
    board.innerHTML=''; board.style.setProperty('--cols',st.length===5?5:st.length>6?4:3);
    st.forEach(function(j,i){
      var d=document.createElement('div'); d.className='jar'+(i===sel?' sel':'')+((j.length===CAP&&j.every(function(x){return x===j[0];}))?' full':''); d.setAttribute('data-i',i);
      var h='<img class="jb" src="img/s/jars_jar_back.webp" alt=""><div class="lys">';
      j.forEach(function(c,k){
        var isTop=k===j.length-1, q=hidden&&!isTop;
        h+='<img class="ly'+(q?' qm':'')+((newJar===i&&k===j.length-1)?' new':'')+'" style="--k:'+k+'" src="img/s/'+(q?'jars_layer_hidden':'jars_layer_'+c)+'.webp" alt="">';
      });
      h+='</div><img class="jf" src="img/s/jars_jar_front.webp" alt="">';
      if(locks[i]!==null){ h+='<img class="lock" src="img/s/jars_lock.webp" alt=""><span class="lockdot" style="--c:'+COL[locks[i]]+'"></span>'; }
      if(j.length===CAP&&j.every(function(x){return x===j[0];})) h+='<img class="done" src="img/s/jars_done.webp" alt="">';
      d.innerHTML=h; board.appendChild(d);
      d.addEventListener('click',function(){ tap(i); });
    });
    $('#m-moves').textContent=moves; $('#m-par').textContent=par;
    var dn=st.filter(function(j){return j.length===CAP&&j.every(function(x){return x===j[0];});}).length, tot=st.filter(function(j){return j.length;}).length;
    $('#m-done').textContent=dn+'/'+Math.max(tot,dn);
  }
  var COL=['#FF5C7A','#FFA13D','#FFD84A','#7ED957','#3FD9A4','#37C8E8','#4C86F0','#9A62E8','#F25FB4','#8B5A3C'];

  function load(i){
    cur=i; var p=P[i]; st=copy(p.jars); locks=p.locks.slice(); hidden=p.hidden; sel=-1; hist=[]; moves=0; par=p.par; won=false; busy=false;
    $('#win').classList.remove('on'); mascot.src='img/s/jars_mascot_idle.webp'; JX.hint('',''); renderPicker(); render();
  }
  function renderPicker(){
    var labels=P.map(function(p,i){ return (i+1)+' · <span class="en i">'+NAMES[i][0]+'</span><span class="tr i">'+NAMES[i][1]+'</span>'; });
    JX.picker($('#picker'),labels,cur,solved,load);
  }
  function tap(i){
    if(busy||won) return;
    if(sel<0){ if(!st[i].length) return; sel=i; render(); return; }
    if(sel===i){ sel=-1; render(); return; }
    var k=canPour(st,sel,i), el=board.children[sel];
    if(!k){ var bad=board.children[i]; sel=-1; render(); var b2=board.children[i]; b2.classList.add('no'); JX.hint('That jar cannot take it: the colours differ, it is locked for another colour, or it is full.','Bu kavanoz alamaz: renkler farklı, başka renge kilitli ya da dolu.'); return; }
    pour(sel,i,k);
  }
  function pour(a,b,k){
    busy=true; hist.push({s:copy(st),m:moves}); JX.hint('','');
    var A=board.children[a], B=board.children[b], ar=A.getBoundingClientRect(), br=B.getBoundingClientRect(), bd=board.getBoundingClientRect();
    var color=COL[top(st[a])], side=(br.left+br.width/2)>=(ar.left+ar.width/2)?1:-1;
    /* kaynak kavanoz hedefin üstüne gelir ve yatar */
    var dx=(br.left+br.width/2-side*br.width*.46)-(ar.left+ar.width/2), dy=(br.top-ar.height*.34)-ar.top;
    A.classList.add('tilt'); A.style.transform='translate('+dx+'px,'+dy+'px) rotate('+(side*62)+'deg)'; mascot.src='img/s/jars_mascot_pour.webp';
    setTimeout(function(){
      var s=document.createElement('i'); s.className='jb-stream'; s.style.setProperty('--c',color);
      s.style.left=(br.left+br.width/2-bd.left)+'px'; s.style.top=(br.top-bd.top+br.height*.05)+'px'; s.style.height=(br.height*.5)+'px'; board.appendChild(s);
      var sp=document.createElement('img'); sp.src='img/s/jars_fx_splash.webp'; sp.className='jb-splash'; sp.alt='';
      sp.style.left=(br.left+br.width/2-bd.left)+'px'; sp.style.top=(br.top-bd.top+br.height*.42)+'px'; setTimeout(function(){ board.appendChild(sp); },240);
      setTimeout(function(){ s.remove(); sp.remove(); },900);
    },300);
    setTimeout(function(){
      var c=top(st[a]); for(var n=0;n<k;n++){ st[a].pop(); st[b].push(c); }
      moves++; sel=-1; render(b); busy=false;
      setTimeout(function(){ if(!won) mascot.src='img/s/jars_mascot_idle.webp'; },500);
      if(isDone(st)) win();
    },820);
  }
  function win(){
    won=true; solved[cur]=true; renderPicker(); mascot.src='img/s/jars_mascot_win.webp'; JX.win(['fx_star','fx_heart','jars_done','jars_fx_drip']);
  }
  $('#b-reset').addEventListener('click',function(){ load(cur); });
  $('#b-undo').addEventListener('click',function(){ if(busy||!hist.length) return; var h=hist.pop(); st=h.s; moves=h.m; sel=-1; won=false; $('#win').classList.remove('on'); render(); });
  $('#b-next').addEventListener('click',function(){ load((cur+1)%P.length); });
  $('#b-hint').addEventListener('click',function(){
    if(busy||won) return; var mv=solve(st);
    if(!mv){ JX.hint('No clear way forward from here. Try Undo or Start over.','Buradan net bir yol görünmüyor. Geri al ya da baştan başla.'); return; }
    sel=-1; render(); var e1=board.children[mv[0]], e2=board.children[mv[1]];
    e1.classList.add('sel'); setTimeout(function(){ e2.style.transform='translateY(-8%)'; },250);
    setTimeout(function(){ e2.style.transform=''; tap2(mv[0],mv[1]); },750);
  });
  function tap2(a,b){ sel=a; tap(b); }
  load(0);
})();
