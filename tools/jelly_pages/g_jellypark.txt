/* ============ Jelly Park: oynanabilir otopark ============
   Kurallar oyunun kendi kodundan (park.dart): bir arabaya dokununca baktığı yönde yol AÇIKSA korna çalıp tahtadan çıkar,
   değilse "bonk"; koniler sabit engeldir, otobüs 3 hücre uzunluğundadır, numaralı arabalar yalnız sırayla çıkar.
   En az dokunuş = araba sayısı. */
(function(){
  'use strict';
  var JG=window.JG,$=JG.$,$$=JG.$$,JX=window.JX;
  var P=window.JX_PUZZLES; if(!P||!P.length) return;
  var board=$('#board'), COLORS=['red','blue','mint','orange','pink','teal','yellow','lilac'];
  var DIRS={u:[-1,0],d:[1,0],l:[0,-1],r:[0,1]}, ROT={u:0,r:90,d:180,l:270};
  var cur=0, pz, n, alive, taps, nextNum, solved=P.map(function(){return false;}), els, won=false, busy=0;
  var NAMES=[['Easy lot','Kolay otopark'],['Cones','Koniler'],['Bus','Otobüs'],['Numbered','Numaralı']];

  function load(i){
    cur=i; pz=P[i]; n=pz.n; alive=pz.cars.map(function(){return true;}); taps=0; nextNum=1; won=false; busy=0;
    $('#win').classList.remove('on'); JX.hint('','');
    JX.picker($('#picker'),P.map(function(p,k){ return '<span class="en i">'+NAMES[k][0]+'</span><span class="tr i">'+NAMES[k][1]+'</span>'; }),cur,solved,load);
    board.innerHTML=''; board.style.setProperty('--n',n);
    for(var r=0;r<n;r++) for(var c=0;c<n;c++){ var t=document.createElement('i'); t.className='pk-tile'; t.style.cssText='left:'+(c/n*100)+'%;top:'+(r/n*100)+'%;background-image:url(img/s/park_tile_'+((r+c)%2?'b':'a')+'.webp)'; board.appendChild(t); }
    pz.cones.forEach(function(q,k){ var d=document.createElement('div'); d.className='pk-cone'; d.style.cssText='left:'+(q[1]/n*100)+'%;top:'+(q[0]/n*100)+'%'; d.innerHTML='<img src="img/s/park_cone_'+(k%2)+'.webp" alt="">'; board.appendChild(d); });
    els=pz.cars.map(function(car,idx){
      var len=car.len, cx=0, cy=0; car.cells.forEach(function(q){ cy+=q[0]+.5; cx+=q[1]+.5; }); cx/=len; cy/=len;
      var b=document.createElement('button'); b.type='button'; b.className='pk-car'; b.setAttribute('aria-label','car '+(idx+1));
      b.style.cssText='width:'+(100/n)+'%;height:'+(100/n*len)+'%;left:'+((cx-.5)/n*100)+'%;top:'+((cy-len/2)/n*100)+'%;transform:rotate('+ROT[car.dir]+'deg);--unrot:'+(-ROT[car.dir])+'deg';
      b.innerHTML='<img src="img/s/park_car'+len+'_'+COLORS[car.color%8]+'.webp" alt="">'+(car.num?'<span class="num"><img src="img/s/park_num_badge.webp" alt=""><b>'+car.num+'</b></span>':'');
      b.addEventListener('click',function(){ tap(idx); }); board.appendChild(b); return b;
    });
    var ex=document.createElement('div'); ex.className='pk-exit'; board.appendChild(ex);
    stats();
  }
  function stats(){ var left=alive.filter(Boolean).length; $('#m-left').textContent=left; $('#m-taps').textContent=taps; $('#m-par').textContent=pz.cars.length; }
  function occ(skip){
    var m={}; pz.cars.forEach(function(car,i){ if(alive[i]&&i!==skip) car.cells.forEach(function(q){ m[q[0]+','+q[1]]=i; }); });
    pz.cones.forEach(function(q){ m[q[0]+','+q[1]]=-1; }); return m;
  }
  function pathFree(i){
    var car=pz.cars[i], d=DIRS[car.dir], m=occ(i), r=car.cells[0][0]+d[0], c=car.cells[0][1]+d[1], steps=0;
    while(r>=0&&r<n&&c>=0&&c<n){ if(m[r+','+c]!==undefined) return -1; r+=d[0]; c+=d[1]; steps++; }
    return steps;
  }
  function tap(i){
    if(won||!alive[i]||els[i].classList.contains('go')) return; taps++;
    var car=pz.cars[i];
    if(car.num && car.num!==nextNum){ bonk(i); JX.hint('Numbered cars leave in order: car '+nextNum+' goes first.','Numaralı arabalar sırayla çıkar: önce '+nextNum+' numaralı araba.'); stats(); return; }
    var steps=pathFree(i);
    if(steps<0){ bonk(i); JX.hint('Bonk! Something is in the way. Try another car.','Bonk! Önünde bir şey var. Başka bir arabayı dene.'); stats(); return; }
    JX.hint('','');
    alive[i]=false; if(car.num) nextNum++;
    var d=DIRS[car.dir], dist=steps+car.len+1, b=els[i]; b.classList.add('go'); b.classList.remove('glow');
    honk(car);
    b.style.transform='translate('+(d[1]*dist*100)+'%,'+(d[0]*dist*100)+'%) rotate('+ROT[car.dir]+'deg)';
    /* dönüş kesin: translate önce, rotate sonra; yüzde eksenler kutuya göre, bu yüzden dönüş sonrası yönü elle çözüyoruz */
    var cellPx=board.getBoundingClientRect().width/n; b.style.transform='translate('+(d[1]*dist*cellPx)+'px,'+(d[0]*dist*cellPx)+'px) rotate('+ROT[car.dir]+'deg)';
    b.style.opacity=0; setTimeout(function(){ b.style.display='none'; },900);
    stats(); if(!alive.some(Boolean)) setTimeout(win,650);
  }
  function bonk(i){ var b=els[i]; b.classList.remove('bonk'); void b.offsetWidth; b.classList.add('bonk'); }
  function honk(car){
    var h=document.createElement('img'); h.src='img/s/park_honk_flash.webp'; h.className='pk-honk'; h.alt='';
    h.style.left=((car.cells[0][1]-.1)/n*100)+'%'; h.style.top=((car.cells[0][0]-.1)/n*100)+'%'; board.appendChild(h); setTimeout(function(){ h.remove(); },650);
  }
  function win(){
    won=true; solved[cur]=true;
    JX.picker($('#picker'),P.map(function(p,k){ return '<span class="en i">'+NAMES[k][0]+'</span><span class="tr i">'+NAMES[k][1]+'</span>'; }),cur,solved,load);
    JX.win(['fx_star','fx_heart','park_honk_flash','park_num_badge']);
  }
  $('#b-reset').addEventListener('click',function(){ load(cur); });
  $('#b-next').addEventListener('click',function(){ load((cur+1)%P.length); });
  $('#b-hint').addEventListener('click',function(){
    if(won) return;
    for(var i=0;i<pz.cars.length;i++){ if(!alive[i]) continue; var car=pz.cars[i]; if(car.num&&car.num!==nextNum) continue; if(pathFree(i)>=0){
      els.forEach(function(e){ e.classList.remove('glow'); }); els[i].classList.add('glow'); setTimeout(function(){ els[i].classList.remove('glow'); },2200);
      JX.hint('This car has a clear road. Tap it!','Bu arabanın yolu açık. Dokun!'); return; } }
  });
  load(0);
})();
