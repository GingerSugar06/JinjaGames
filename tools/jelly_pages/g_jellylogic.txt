/* ============ Jelly Logic: oynanabilir sudoku ============
   Kurallar oyunun kendi kodundan (logic.dart): satır, sütun ve kutu başına her sayı bir kez; karşılaştırma okları KÜÇÜK sayıyı
   gösterir; kare hücre TEK, daire hücre ÇİFT sayı taşır; çapraz sudokuda iki çapraz da her sayıyı bir kez içerir.
   Yanlış sayı hemen kırmızı olur (çözümle karşılaştırılır). Bulmacalar tek çözümlü. */
(function(){
  'use strict';
  var JG=window.JG,$=JG.$,$$=JG.$$,JX=window.JX;
  var P=window.JX_PUZZLES; if(!P||!P.length) return;
  var board=$('#board'), pad=$('#pad'), owl=$('#lg-owl');
  var cur=0, pz, n, val, notes, sel=-1, err=0, noteMode=false, won=false, solved=P.map(function(){return false;}), hinted=-1;
  var NAMES=[['4×4 Sudoku','4×4 Sudoku'],['Bigger or smaller','Büyük ya da küçük'],['Odd & even','Tek ve çift'],['Diagonal','Çapraz'],['6×6 Sudoku','6×6 Sudoku']];

  function load(i){
    cur=i; pz=P[i]; n=pz.n; val=pz.givens.slice(); notes=val.map(function(){return {};}); sel=-1; err=0; won=false; hinted=-1;
    $('#win').classList.remove('on'); owl.src='img/s/logic_owl_idle.webp'; noteMode=false; $('#b-note').classList.remove('on'); JX.hint('','');
    var lab=P.map(function(p,k){ return '<span class="en i">'+NAMES[k][0]+'</span><span class="tr i">'+NAMES[k][1]+'</span>'; });
    JX.picker($('#picker'),lab,cur,solved,load);
    build(); render();
  }
  function build(){
    board.style.setProperty('--n',n); board.innerHTML='';
    for(var i=0;i<n*n;i++){
      var b=document.createElement('button'); b.type='button'; b.className='lc'; b.setAttribute('data-i',i); b.setAttribute('aria-label','cell '+(i+1));
      (function(i){ b.addEventListener('click',function(){ pick(i); }); })(i); board.appendChild(b);
    }
    /* kutu çizgileri */
    for(var c=pz.bc;c<n;c+=pz.bc){ var v=document.createElement('i'); v.className='lg-line'; v.style.cssText='top:2px;bottom:2px;width:4px;left:calc('+(c/n*100)+'% - 2px)'; board.appendChild(v); }
    for(var r=pz.br;r<n;r+=pz.br){ var h=document.createElement('i'); h.className='lg-line'; h.style.cssText='left:2px;right:2px;height:4px;top:calc('+(r/n*100)+'% - 2px)'; board.appendChild(h); }
    /* karşılaştırma okları: ok KÜÇÜK olanı gösterir (a < b) */
    pz.cmp.forEach(function(p){
      var a=p[0], b=p[1], ra=Math.floor(a/n), ca=a%n, rb=Math.floor(b/n), cb=b%n, im=document.createElement('img');
      var dir=rb===ra ? (cb>ca?'left':'right') : (rb>ra?'up':'down');
      im.src='img/s/logic_cmp_'+dir+'.webp'; im.className='lg-cmp'; im.alt='';
      im.style.left=((ca+cb+1)/2/n*100)+'%'; im.style.top=((ra+rb+1)/2/n*100)+'%'; board.appendChild(im);
    });
    pad.style.setProperty('--n',n); pad.innerHTML='';
    for(var k=1;k<=n;k++){ var bt=document.createElement('button'); bt.type='button'; bt.textContent=k; (function(k){ bt.addEventListener('click',function(){ enter(k); }); })(k); pad.appendChild(bt); }
  }
  function onDiag(i){ var r=Math.floor(i/n), c=i%n; return pz.diag && (r===c || r+c===n-1); }
  function peers(i){
    var r=Math.floor(i/n), c=i%n, set={}; for(var k=0;k<n;k++){ set[r*n+k]=1; set[k*n+c]=1; }
    var R=Math.floor(r/pz.br)*pz.br, C=Math.floor(c/pz.bc)*pz.bc; for(var a=0;a<pz.br;a++) for(var b=0;b<pz.bc;b++) set[(R+a)*n+C+b]=1;
    delete set[i]; return set;
  }
  function render(){
    var cells=$$('.lc',board), filled=0, selv=sel>=0?val[sel]:0, counts={};
    cells.forEach(function(el,i){
      var v=val[i], given=pz.givens[i]>0, bad=v&&v!==pz.solution[i];
      el.className='lc'+(given?' gv':'')+(onDiag(i)?' dg':'')+(v&&selv&&v===selv&&i!==sel?' sm':'')+(bad?' er':'')+(i===hinted?' hn':'')+(i===sel?' sel':'');
      var h='';
      if(v){ h+=v; counts[v]=(counts[v]||0)+1; if(!bad) filled++; }
      else { var nt=notes[i], keys=Object.keys(nt); if(keys.length){ h+='<span class="nt">'; for(var k=1;k<=Math.max(n,4);k++){ h+='<i>'+(nt[k]?k:'')+'</i>'; } h+='</span>'; } }
      var pr=pz.parity[i]; if(pr) h+='<span class="pr '+pr+'"></span>';
      el.innerHTML=h;
    });
    $$('button',pad).forEach(function(b,k){ b.classList.toggle('dn',(counts[k+1]||0)>=n); });
    $('#m-fill').textContent=filled+'/'+(n*n); $('#m-err').textContent=err;
    if(!won && filled===n*n) finish();
  }
  function pick(i){ if(won) return; sel=sel===i?-1:i; hinted=-1; render(); }
  function enter(k){
    if(won||sel<0){ if(!won) JX.hint('Tap a cell first.','Önce bir hücreye dokun.'); return; }
    if(pz.givens[sel]) return;
    if(noteMode){ if(val[sel]) return; if(notes[sel][k]) delete notes[sel][k]; else notes[sel][k]=1; render(); return; }
    var wasBad=val[sel]&&val[sel]!==pz.solution[sel];
    val[sel]=(val[sel]===k)?0:k; notes[sel]={};
    if(val[sel] && val[sel]!==pz.solution[sel]){ err++; owl.src='img/s/logic_owl_sad.webp'; shake(); setTimeout(function(){ if(!won) owl.src='img/s/logic_owl_idle.webp'; },900); }
    else if(val[sel]){ owl.src='img/s/logic_owl_wink.webp'; setTimeout(function(){ if(!won) owl.src='img/s/logic_owl_idle.webp'; },700); }
    render();
  }
  function shake(){ var el=$$('.lc',board)[sel]; if(el){ el.classList.remove('shake'); void el.offsetWidth; el.classList.add('shake'); } }
  function finish(){
    won=true; solved[cur]=true; sel=-1; owl.src='img/s/logic_owl_win.webp'; render();
    JX.picker($('#picker'),P.map(function(p,k){ return '<span class="en i">'+NAMES[k][0]+'</span><span class="tr i">'+NAMES[k][1]+'</span>'; }),cur,solved,load);
    JX.win(['fx_star','fx_heart','logic_fx_ok','logic_fx_note']);
  }
  $('#b-note').addEventListener('click',function(){ noteMode=!noteMode; this.classList.toggle('on',noteMode); });
  $('#b-erase').addEventListener('click',function(){ if(won||sel<0||pz.givens[sel]) return; val[sel]=0; notes[sel]={}; render(); });
  $('#b-reset').addEventListener('click',function(){ load(cur); });
  $('#b-next').addEventListener('click',function(){ load((cur+1)%P.length); });
  $('#b-hint').addEventListener('click',function(){
    if(won) return;
    var pick1=-1; for(var i=0;i<n*n;i++){ if(val[i] && val[i]!==pz.solution[i]){ pick1=i; break; } }
    if(pick1<0){ var best=-1, bc=99; for(var j=0;j<n*n;j++){ if(!val[j]){ var pe=peers(j), used={}; Object.keys(pe).forEach(function(q){ if(val[q]) used[val[q]]=1; }); var c=n-Object.keys(used).length; if(c<bc){ bc=c; best=j; } } } pick1=best; }
    if(pick1<0) return;
    val[pick1]=pz.solution[pick1]; notes[pick1]={}; hinted=pick1; sel=pick1;
    JX.hint('The hint fills one cell. Look at its row, column and box to see why it fits.','İpucu bir hücreyi doldurur. Neden uyduğunu görmek için satırına, sütununa ve kutusuna bak.'); render();
  });
  document.addEventListener('keydown',function(e){
    if(e.target.closest&&e.target.closest('input,textarea')) return;
    var k=parseInt(e.key,10); var r=board.getBoundingClientRect(); if(r.bottom<0||r.top>innerHeight) return;
    if(k>=1&&k<=n) enter(k);
  });
  load(0);
})();
